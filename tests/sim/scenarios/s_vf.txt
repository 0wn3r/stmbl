# plant carries the alpha 0.5 estimate's flux (the V/f config has none).
title Spindle V/f open loop, 0 -> 838 -> 0 rad/s
config conf/fanuc_a05_spindle_vf.txt
plant lmr=0.0216 tr=0.04 i_n=5.1 f=0.06 d=0.00016
f4 vf0.vel_cmd = 0
at 0.1 f4 fault0.en = 1
at 0.2 f4 vf0.vel_cmd = 838
at 10.0 f4 vf0.vel_cmd = 0
trace plant.vel vf0.vel vf0.stall hv0.id_fb hv0.iq_fb plant.i_mr vf0.u_cmd vf0.slip
expect max fault0.last_fault < 0.5 from 0.15
expect final plant.vel > 825 from 9.9 to 10.0
expect max hv0.iq_fb < 15.5
expect absmax plant.vel < 5 from 19.5
run 19.6
