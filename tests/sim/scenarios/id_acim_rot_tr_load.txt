# idacim rotating tr check at three loads (rot_tr_loads 1) against a plant
# whose slip grows with load like acim_flux0.tr_load 0.1, on the spindle's
# FOC config. 29 Oct: rot_tr_load 0.107 (plant tr_load 0.1), 0.008 (0),
# 0.284 (0.3). Levels 0.6 / 0.45 / 0.3 of id_n on q give iq/i_mr 1.53 /
# 1.25 / 0.91; a level of 1.0 runs the window through faster than the 3 tr
# wait, one of 0.2 does not beat the plant's drag.
title idacim rotating tr check at three loads
config conf/fanuc_a2_spindle_foc_enc.txt
f4 link id_acim
f4 fault0.en = 0
f4 idacim0.test_cur = 16
f4 idacim0.n_volt = 155
f4 idacim0.n_freq = 78.4
f4 idacim0.n_cur = 19
f4 idacim0.n_pp = 2
f4 idacim0.spin = 1
f4 idacim0.rot_tr_vel = 600
f4 idacim0.rot_tr_loads = 1
plant r=0.19 ld=0.0008 lmr=0.0124 tr=0.12 i_knee=0 lmr_sat=0 tr_sat=0 i_dip=0 lmr_dip=0 tr_dip=0 tr_load=0.1 jm=0.0118 f=0.5 d=0.0013 dt_shape=1 dt_i0=0.85 dt_eff=1.77e-6
at 0.1 f4 fault0.en = 1
at 0.2 f4 idacim0.state = 1.2
at 21 f4 idacim0.state = 4.2
trace idacim0.state hv0.id_fb hv0.iq_fb plant.vel plant.i_mr idacim0.d_cmd idacim0.q_cmd
expect max fault0.last_fault < 0.5 from 0.15
expect final idacim0.tr > 0
expect final idacim0.rot_tr > 0
expect final idacim0.rot_tr_k > 0
expect final idacim0.rot_tr_imr > 0
expect final idacim0.rot_tr_lmr > 0
expect final idacim0.rot_tr_b > -1
expect final idacim0.rot_tr_load > 0.08
expect final idacim0.rot_tr_load < 0.12
run 220
